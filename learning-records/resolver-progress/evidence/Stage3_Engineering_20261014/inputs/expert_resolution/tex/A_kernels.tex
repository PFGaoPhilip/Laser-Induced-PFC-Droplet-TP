\section{Axisymmetric kernels (derivation)}\label{app:kernels}
\emph{Symbols.} Field point $\mathbf x=(r,0,z)$ and source ring point
$\mathbf y=(r'\cos\theta,r'\sin\theta,z')$ have length coordinates (m); primes label the source,
not derivatives. The unit normal $\mathbf n=(n_r\cos\theta,n_r\sin\theta,n_z)$ and azimuth
$\theta$ (radians) are dimensionless.
Define $A_{\mathrm K}=r^2+r'^2+(z-z')^2$, $B_{\mathrm K}=2rr'$ (m$^2$),
$m=2B_{\mathrm K}/(A_{\mathrm K}+B_{\mathrm K})\in[0,1)$ (dimensionless);
$\KE(m),\EE(m)$ are the dimensionless complete elliptic integrals of the first and second kind.
Their definitions are the integrals of $(1-m\sin^2\varphi)^{-1/2}$ and
$(1-m\sin^2\varphi)^{1/2}$ over $\varphi\in[0,\pi/2]$.
The formulas below also hold in the rescaled coordinates of the BIM. Distinct noncoincident
points require $A_{\mathrm K}>B_{\mathrm K}\ge0$.

\textbf{Step K1.} $|\mathbf x-\mathbf y|^2=A_{\mathrm K}-B_{\mathrm K}\cos\theta$.
Substituting $\theta=\pi-2\varphi$ gives $\cos\theta=2\sin^2\varphi-1$, so the denominator becomes
$(A_{\mathrm K}+B_{\mathrm K})(1-m\sin^2\varphi)$ and $\dd\theta=-2\dd\varphi$.
The transformed limits are $\varphi=\pi/2,-\pi/2$; reversing them removes the minus sign and
evenness doubles the half-interval integral. Immediately below, $A_{\mathrm K},B_{\mathrm K}$
are squared lengths, $\theta$ is the dummy angle, and $m,\KE$ the dimensionless parameter/function above:
\begin{equation}
\int_0^{2\pi}\frac{\dd\theta}{(A_{\mathrm K}-B_{\mathrm K}\cos\theta)^{1/2}}
=\frac{4\KE(m)}{\sqrt{A_{\mathrm K}+B_{\mathrm K}}} .\label{eq:K1}
\end{equation}
\textbf{Step K2.} The auxiliary identity follows by differentiating
$F(\varphi)=\sin\varphi\cos\varphi/(1-m\sin^2\varphi)^{1/2}$:
$F'=(1-2\sin^2\varphi+m\sin^4\varphi)/(1-m\sin^2\varphi)^{3/2}$.
Subtracting the elliptic-$E$ integrand from $(1-m)(1-m\sin^2\varphi)^{-3/2}$ gives $-mF'$;
its integral is zero because $F(0)=F(\pi/2)=0$. Thus
$\int_0^{\pi/2}(1-m\sin^2\varphi)^{-3/2}\dd\varphi=\EE(m)/(1-m)$.
Substitute this identity
and $1-m=(A_{\mathrm K}-B_{\mathrm K})/(A_{\mathrm K}+B_{\mathrm K})$, define
$I_0$ as the inverse-cube angular integral (m$^{-3}$). Here $\theta$ and $m,\EE$ are
dimensionless; $A_{\mathrm K}>B_{\mathrm K}\ge0$ are squared lengths:
\begin{equation}
I_0\coloneqq\int_0^{2\pi}\frac{\dd\theta}{(A_{\mathrm K}-B_{\mathrm K}\cos\theta)^{3/2}}
=\frac{4\EE(m)}{(A_{\mathrm K}-B_{\mathrm K})\sqrt{A_{\mathrm K}+B_{\mathrm K}}} .\label{eq:K2}
\end{equation}
\textbf{Step K3.} For $B_{\mathrm K}>0$, write
$\cos\theta=[A_{\mathrm K}-(A_{\mathrm K}-B_{\mathrm K}\cos\theta)]/B_{\mathrm K}$ and use
Eq.~\eqref{eq:K1} and Eq.~\eqref{eq:K2}. Before the next identity,
$I_0,I_1$ have unit m$^{-3}$, $A_{\mathrm K},B_{\mathrm K}$ m$^2$, $\theta$ is the dummy angle,
and $\KE=\KE(m),\EE=\EE(m)$ are dimensionless. The $B_{\mathrm K}=0$ axis limit is handled separately:
\begin{align}
I_1\coloneqq\int_0^{2\pi}\frac{\cos\theta\,\dd\theta}
{(A_{\mathrm K}-B_{\mathrm K}\cos\theta)^{3/2}}
&=\frac{A_{\mathrm K}I_0}{B_{\mathrm K}}-\frac{4\KE}{B_{\mathrm K}\sqrt{A_{\mathrm K}+B_{\mathrm K}}}\notag\\
&=\frac{4}{B_{\mathrm K}\sqrt{A_{\mathrm K}+B_{\mathrm K}}}
\Bigl[\frac{A_{\mathrm K}\EE}{A_{\mathrm K}-B_{\mathrm K}}-\KE\Bigr].\label{eq:K3}
\end{align}
For $B_{\mathrm K}=0$, the integral of $\cos\theta$ vanishes, so $I_1=0$ directly.
For $m<10^{-5}$ the binomial series avoids cancellation:
$I_1=\tfrac{3\pi}{2}B_{\mathrm K}A_{\mathrm K}^{-5/2}
[1+\tfrac{35}{32}(B_{\mathrm K}/A_{\mathrm K})^2+O(B_{\mathrm K}^4/A_{\mathrm K}^4)]$ (from
$(1-\epsilon\cos\theta)^{-3/2}=1+\tfrac32\epsilon\cos\theta+\tfrac{15}{8}\epsilon^2\cos^2\theta+\tfrac{35}{16}\epsilon^3\cos^3\theta+\dots$,
and $\int_0^{2\pi}\cos^2\theta\,\dd\theta=\pi$,
$\int_0^{2\pi}\cos^4\theta\,\dd\theta=3\pi/4$, with
$\epsilon=B_{\mathrm K}/A_{\mathrm K}$ dimensionless).
\textbf{Step K4.} Before the single/double layer kernels, $g_{\mathrm K}$ is dimensionless,
$h_{\mathrm K}$ has unit m$^{-1}$; $r,r',z,z',\mathbf x,\mathbf y$ are lengths,
$A_{\mathrm K},B_{\mathrm K}$ squared lengths, $I_0,I_1$ inverse-cube lengths,
$n_r,n_z$ dimensionless normal components, $\KE$ the elliptic function, and $\theta$ the dummy angle:
\begin{align}
g_{\mathrm K}=r'\!\int_0^{2\pi}\!\frac{\dd\theta}{|\mathbf x-\mathbf y|}
&=\frac{4r'\KE}{\sqrt{A_{\mathrm K}+B_{\mathrm K}}},\notag\\
h_{\mathrm K}=r'\!\int_0^{2\pi}\!\frac{(\mathbf x-\mathbf y)\cdot\mathbf n}{|\mathbf x-\mathbf y|^3}\dd\theta
&=r'\bigl[n_r(rI_1-r'I_0)+n_z(z-z')I_0\bigr],\label{eq:K4}
\end{align}
since $(\mathbf x-\mathbf y)\cdot\mathbf n=n_r(r\cos\theta-r')+n_z(z-z')$. The image terms use $z'\to-z'$, $n_z\to-n_z$.
\textbf{Step K5 (singularity).} As $\mathbf y\to\mathbf x$ on the surface,
$\epsilon_m\coloneqq1-m=[(r-r')^2+(z-z')^2]/(A_{\mathrm K}+B_{\mathrm K})\to0$ and
$\KE=P(\epsilon_m)-Q(\epsilon_m)\ln\epsilon_m$ (Abramowitz \& Stegun 17.3.34,
dimensionless polynomials $P,Q$, error $\le2\times10^{-8}$). On an element
adjacent to the collocation node, with $\xi$ the dimensionless parametric distance from the node,
$-Q\ln\epsilon_m=-Q\ln(\epsilon_m/\xi^2)+2Q\ln(1/\xi)$; the first part is smooth (Gauss) and the second is integrated with the
8-point Gauss rule for weight $\ln(1/\xi)$. All kernels were checked against direct $10^5$-point azimuthal quadrature
(maximum relative error $1.2\times10^{-14}$).
